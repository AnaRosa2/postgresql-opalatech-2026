Query executada 
SELECT
    h.nome, q.numero, q.tipo,
    r.data_entrada, r.data_saida, r.valor_total
FROM reservas r
JOIN hospedes h ON r.hospede_id = h.id
JOIN quartos q  ON r.numero_quarto = q.numero
ORDER BY r.data_entrada DESC;
